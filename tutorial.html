Tic Tac Toe Game Tutorial

Leer hoe je dit klassieke spel maakt met HTML, CSS en JavaScript

🚀 Aan de slag

Volg deze stappen om je ontwikkelomgeving klaar te maken:

1. Projectmap aanmaken

  1. Open Verkenner (Windows + E)
  2. Navigeer naar waar je je project wilt opslaan (bv. Documenten)
  3. Klik met rechts → Nieuw → Map
  4. Geef de map een naam zoals "tic-tac-toe"

2. Bestanden aanmaken

  1. Open de nieuwe map in Verkenner
  2. Klik met rechts → Nieuw → Tekstdocument
  3. Hernoem het bestand naar index.html
  4. Herhaal dit voor style.css en script.js
  5. Klik "Ja" als Windows waarschuwt over de bestandsextensie

3. Openen in VSCode

  1. Open Visual Studio Code
  2. Klik op "Bestand" → "Map openen"
  3. Selecteer je "tic-tac-toe" map
  4. Je ziet nu je bestanden in de linker sidebar
Tip: Dubbelklik op een bestand in VSCode om het te bewerken.

JavaScript Spellogica

JavaScript regelt alle spelmechanismen:

const cells = document.querySelectorAll(".cell");
const currentPlayerSpan = document.getElementById("current-player");
const winnerSpan = document.getElementById("winner");
const winnerText = document.querySelector(".winner");
const resetButton = document.querySelector(".reset-button");

let currentPlayer = "X";
let board = ["", "", "", "", "", "", "", "", ""];
let gameActive = true;

// Winnende combinaties
const winningCombinations = [
    [0, 1, 2], [3, 4, 5], [6, 7, 8], // rijen
    [0, 3, 6], [1, 4, 7], [2, 5, 8], // kolommen
    [0, 4, 8], [2, 4, 6]             // diagonalen
];

// Celklik afhandelen
function handleCellClick(event) {
    const cell = event.target;
    const cellIndex = Array.from(cells).indexOf(cell);

    if (board[cellIndex] !== "" || !gameActive) return;

    board[cellIndex] = currentPlayer;
    cell.textContent = currentPlayer;

    if (checkWinner()) {
        gameActive = false;
        winnerSpan.textContent = currentPlayer;
        winnerText.style.display = "block";
        return;
    }

    if (board.every((cell) => cell !== "")) {
        gameActive = false;
        winnerSpan.textContent = "Gelijkspel";
        winnerText.style.display = "block";
        return;
    }

    currentPlayer = currentPlayer === "X" ? "O" : "X";
    currentPlayerSpan.textContent = currentPlayer;
}

// Winnaar controleren
function checkWinner() {
    return winningCombinations.some((combination) => {
        return combination.every((index) => board[index] === currentPlayer);
    });
}

// Spel resetten
function resetGame() {
    board = ["", "", "", "", "", "", "", "", ""];
    gameActive = true;
    currentPlayer = "X";
    currentPlayerSpan.textContent = currentPlayer;
    winnerSpan.textContent = "";
    winnerText.style.display = "none";
    cells.forEach((cell) => (cell.textContent = ""));
}

// Event listeners
cells.forEach((cell) => cell.addEventListener("click", handleCellClick));
resetButton.addEventListener("click", resetGame);

Belangrijke onderdelen:

  • Spelstatus: Volgt de huidige speler en bord
  • Wincontrole: Checkt alle winnende combinaties
  • Beurtwissel: Wisselt tussen X en O
